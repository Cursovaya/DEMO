CREATE DATABASE shoes_store
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE shoes_store;

-- =====================================================
-- 1. ТАБЛИЦА ПОЛЬЗОВАТЕЛЕЙ (Users)
-- =====================================================
CREATE TABLE Users (
    UserID INT PRIMARY KEY AUTO_INCREMENT,
    Login VARCHAR(100) NOT NULL UNIQUE,
    Password VARCHAR(255) NOT NULL,
    FullName VARCHAR(255),
    Role VARCHAR(50) DEFAULT 'client'
);

-- =====================================================
-- 2. ТАБЛИЦА КАТЕГОРИЙ (Categories)
-- =====================================================
CREATE TABLE Categories (
    CategoryID INT PRIMARY KEY AUTO_INCREMENT,
    Name VARCHAR(100) NOT NULL UNIQUE
);

-- =====================================================
-- 3. ТАБЛИЦА ПРОИЗВОДИТЕЛЕЙ (Manufacturers)
-- =====================================================
CREATE TABLE Manufacturers (
    ManufacturerID INT PRIMARY KEY AUTO_INCREMENT,
    Name VARCHAR(100) NOT NULL UNIQUE
);

-- =====================================================
-- 4. ТАБЛИЦА ПОСТАВЩИКОВ (Suppliers)
-- =====================================================
CREATE TABLE Suppliers (
    SupplierID INT PRIMARY KEY AUTO_INCREMENT,
    Name VARCHAR(100) NOT NULL UNIQUE
);

-- =====================================================
-- 5. ТАБЛИЦА ТОВАРОВ (Products)
-- =====================================================
CREATE TABLE Products (
    ProductID INT PRIMARY KEY AUTO_INCREMENT,
    Name VARCHAR(100) NOT NULL,
    CategoryID INT,
    ManufacturerID INT,
    SupplierID INT,
    Price DECIMAL(10, 2) NOT NULL DEFAULT 0,
    StockQuantity INT NOT NULL DEFAULT 0,
    Unit VARCHAR(10) DEFAULT 'шт',
    Discount DECIMAL(5, 2) DEFAULT 0.00,
    Description TEXT,
    ImagePath VARCHAR(255),
    FOREIGN KEY (CategoryID) REFERENCES Categories(CategoryID),
    FOREIGN KEY (ManufacturerID) REFERENCES Manufacturers(ManufacturerID),
    FOREIGN KEY (SupplierID) REFERENCES Suppliers(SupplierID)
);

-- =====================================================
-- 6. ТАБЛИЦА СТАТУСОВ ЗАКАЗОВ (не используется напрямую, но для данных)
-- =====================================================
CREATE TABLE OrderStatuses (
    StatusID INT PRIMARY KEY AUTO_INCREMENT,
    Name VARCHAR(50) NOT NULL UNIQUE
);

-- =====================================================
-- 7. ТАБЛИЦА ЗАКАЗОВ (Orders)
-- =====================================================
CREATE TABLE Orders (
    OrderID INT PRIMARY KEY AUTO_INCREMENT,
    UserID INT,
    Status VARCHAR(50) DEFAULT 'Новый',
    DeliveryAddress VARCHAR(255),
    OrderDate DATE DEFAULT (CURRENT_DATE),
    IssueDate DATE,
    FOREIGN KEY (UserID) REFERENCES Users(UserID)
);

-- =====================================================
-- 8. ТАБЛИЦА ТОВАРОВ В ЗАКАЗЕ (OrderItems)
-- =====================================================
CREATE TABLE OrderItems (
    ItemID INT PRIMARY KEY AUTO_INCREMENT,
    OrderID INT,
    ProductID INT,
    Quantity INT NOT NULL,
    FOREIGN KEY (OrderID) REFERENCES Orders(OrderID) ON DELETE CASCADE,
    FOREIGN KEY (ProductID) REFERENCES Products(ProductID)
);
